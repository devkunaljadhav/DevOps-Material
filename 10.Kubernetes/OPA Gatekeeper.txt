#OPA Gatekeeper

#Our practical goal: Install OPA Gatekeeper in Kubernetes and enforce a policy that every Pod must have an env label.

#Check nodes      
kubectl get nodes

#Check Kubernetes connection     
kubectl cluster-info

#Check Helm 
helm version       

#Install Helm
curl -fsSL -o get_helm.sh \
https://raw.githubusercontent.com/helm/helm/main/scripts/get-helm-3

chmod 700 get_helm.sh

./get_helm.sh


#Install Gatekeeper
helm repo add gatekeeper \
https://open-policy-agent.github.io/gatekeeper/charts

helm repo update
helm install gatekeeper gatekeeper/gatekeeper \
--namespace gatekeeper-system \
--create-namespace


#Verify installation
kubectl get pods -n gatekeeper-system


#Check services
kubectl get svc -n gatekeeper-system

#Check CRDs
kubectl get crd | grep gatekeeper


#Create a policy that checks whether every Pod has an env label.

vi require-env-template.yml


apiVersion: templates.gatekeeper.sh/v1
kind: ConstraintTemplate
metadata:
  name: k8srequiredenvlabel
spec:
  crd:
    spec:
      names:
        kind: K8sRequiredEnvLabel
      validation:
        openAPIV3Schema:
          type: object
          properties:
            message:
              type: string

  targets:
    - target: admission.k8s.gatekeeper.sh
      rego: |
        package k8srequiredenvlabel

        violation[{"msg": msg}] {
          labels := object.get(
            input.review.object.metadata,
            "labels",
            {}
          )

          not labels.env

          msg := "Pod must have an env label (env=dev or env=prod)"
        }


kubectl apply -f require-env-template.yml


#Create Constraint YAML

vi require-env-constraint.yml


apiVersion: constraints.gatekeeper.sh/v1beta1
kind: K8sRequiredEnvLabel
metadata:
  name: require-env-label
spec:
  enforcementAction: deny

  match:
    kinds:
      - apiGroups: [""]
        kinds:
          - Pod


kubectl apply -f require-env-constraint.yml

kubectl get constraints


#Create Pod without env label
kubectl run testpod --image=nginx

#Expected result:
Error from server:
admission webhook "validation.gatekeeper.sh" denied the request:

[require-env-label]
Pod must have an env label
(env=dev or env=prod)



kubectl get pods


#Create Pod with env=dev

kubectl run testpod --image=nginx \
--labels=env=dev

#Expected:
pod/testpod created
